            import json
            # Connessione al server
            io = remote('socket.cryptohack.org', 11112)
            # Riceviamo il testo iniziale per "pulire" la comunicazione
            # recvuntil è più sicuro: legge tutto finché non trova la stringa indicata
            io.recvuntil(b"I hope that's ok.\n\n")
            # Prepariamo il messaggio JSON
            # Il server vuole {"buy": "flag"}
            payload = {"buy": "flag"}
            # Inviamo il payload convertito in stringa JSON
            io.sendline(json.dumps(payload).encode())
            # Riceviamo e stampiamo la risposta finale
            # Questa riga conterrà la flag!
            flag_data = io.recvline()
            print("--- RISULTATO ---")
            print(flag_data.decode())
            print("-----------------")
            # Chiudiamo la connessione
            io.close()
