from flask import Flask, render_template, jsonify, request
import json
import os

app = Flask(__name__)

# File che fungerà da database dinamico per il server
DB_FILE = 'server_db.json'

def init_db():
    """Inizializza il database se non esiste"""
    if not os.path.exists(DB_FILE):
        default_data = {
            "users": {
                'admin': {'pass': 'admin999', 'role': 'admin', 'status': 'active', 'perms': 'Level 5'},
                'user': {'pass': 'password123', 'role': 'user', 'status': 'active', 'perms': 'Level 1'}
            },
            "logs": []
        }
        with open(DB_FILE, 'w') as f:
            json.dump(default_data, f)

def load_db():
    """Carica i dati dal file JSON"""
    with open(DB_FILE, 'r') as f:
        return json.load(f)

def save_db(data):
    """Salva i dati nel file JSON"""
    with open(DB_FILE, 'w') as f:
        json.dump(data, f)

# Inizializza il DB all'avvio del server
init_db()

@app.route('/')
def index():
    """Rotte principale che serve la tua interfaccia HTML"""
    return render_template('index.html')

@app.route('/api/data', methods=['GET', 'POST'])
def handle_data():
    """API REST per sincronizzare il frontend con il backend Flask"""
    if request.method == 'POST':
        # Quando il frontend invia nuovi dati (nuovi utenti, log, ecc.), li salviamo
        data = request.json
        save_db(data)
        return jsonify({"status": "success"})
    else:
        # Quando il frontend richiede i dati all'avvio
        return jsonify(load_db())

@app.route('/api/login', methods=['POST'])
def login():
    """Endpoint per verificare le credenziali (SIMULAZIONE VULNERABILITÀ)"""
    user = request.form.get('username', '')
    pwd = request.form.get('password', '')
    db = load_db()
    
    # --- SIMULAZIONE VULNERABILITA' AVANZATA ---
    # Uniamo username e password in maiuscolo per analizzare il payload simulato
    payload = (user + " " + pwd).upper()

    # 1. Simulazione Data Exfiltration: Tabella Users
    # Payload d'esempio: ' UNION SELECT username, password FROM users --
    if "UNION" in payload and "SELECT" in payload and "USERS" in payload:
        return jsonify({
            "status": "sqli_dump",
            "table": "users",
            "cols": ["username", "pass", "role", "perms", "status"],
            "data": db.get('users', {})
        })

    # 2. Simulazione Data Exfiltration: Tabella Logs
    # Payload d'esempio: ' UNION SELECT time, user, pass FROM logs --
    elif "UNION" in payload and "SELECT" in payload and "LOGS" in payload:
        return jsonify({
            "status": "sqli_dump",
            "table": "access_logs",
            "cols": ["time", "user", "pass", "event", "status"],
            "data": db.get('logs', [])
        })

    # 3. Simulazione Authentication Bypass Classico
    # Payload d'esempio: admin' OR 1=1 --
    elif "'" in pwd or "'" in user or "%27" in pwd or "%27" in user:
        return jsonify({"status": "sqli_bypass", "user": "admin", "role": "admin"})
            
    # --- AUTENTICAZIONE STANDARD ---
    # Controllo di sicurezza: se il database si è corrotto, lo ripariamo al volo
    if 'users' not in db:
        repaired_data = {
            "users": {
                'admin': {'pass': 'admin999', 'role': 'admin', 'status': 'active', 'perms': 'Level 5'},
                'user': {'pass': 'password123', 'role': 'user', 'status': 'active', 'perms': 'Level 1'}
            },
            "logs": db.get("logs", [])
        }
        save_db(repaired_data)
        db = repaired_data

    # Verifica finale delle credenziali
    if user in db['users'] and db['users'][user]['pass'] == pwd:
        user_data = db['users'][user]
        
        # RISOLUZIONE PROBLEMA: Se l'utente ha "Level 5", forziamo l'interfaccia Admin
        if user_data.get('perms') == 'Level 5':
            ruolo_interfaccia = 'admin'
        else:
            ruolo_interfaccia = user_data.get('role', 'user')
            
        return jsonify({"status": "success", "user": user, "role": ruolo_interfaccia})
        
    return jsonify({"status": "failure", "message": "Credenziali errate"}), 401

if __name__ == '__main__':
    # Flask gira sulla 5020, il WAF sulla 32456
    app.run(host='0.0.0.0', port=5020, debug=True)
